cmake_minimum_required(VERSION 3.16)
project(hector_backend_core VERSION 0.6.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Ceres REQUIRED)
find_package(Boost REQUIRED COMPONENTS iostreams)
find_package(absl REQUIRED)
find_package(Protobuf REQUIRED)
find_package(glog REQUIRED)
find_package(gflags REQUIRED)
find_package(Lua 5.3 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CAIRO REQUIRED cairo)
set(VENDOR ${CMAKE_CURRENT_SOURCE_DIR}/../3rd/cartographer)
set(CARTOGRAPHER_CONFIGURATION_FILES_DIRECTORY ${CMAKE_INSTALL_PREFIX}/share/${PROJECT_NAME}/configuration_files)
set(CARTOGRAPHER_HAS_GRPC OFF)
configure_file(${VENDOR}/cartographer/common/config.h.cmake ${CMAKE_CURRENT_BINARY_DIR}/cartographer/common/config.h)
file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS ${VENDOR}/cartographer/*.cc)
list(FILTER SOURCES EXCLUDE REGEX "/cloud/|_test\\.cc$|_main\\.cc$|test_helpers|/mock_|/fake_")
file(GLOB_RECURSE PROTOS ${VENDOR}/cartographer/*.proto)
list(FILTER PROTOS EXCLUDE REGEX "/cloud/")
foreach(proto ${PROTOS})
 file(RELATIVE_PATH rel ${VENDOR} ${proto})
 string(REGEX REPLACE "\\.proto$" ".pb.cc" cc ${rel})
 string(REGEX REPLACE "\\.proto$" ".pb.h" hh ${rel})
 add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/${cc} ${CMAKE_CURRENT_BINARY_DIR}/${hh}
 COMMAND protobuf::protoc --cpp_out=${CMAKE_CURRENT_BINARY_DIR} -I${VENDOR} ${proto} DEPENDS ${proto} VERBATIM)
 list(APPEND SOURCES ${CMAKE_CURRENT_BINARY_DIR}/${cc})
endforeach()
add_library(${PROJECT_NAME} STATIC ${SOURCES})
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${VENDOR}> $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}> $<INSTALL_INTERFACE:include> ${LUA_INCLUDE_DIR} ${CAIRO_INCLUDE_DIRS})
target_link_libraries(${PROJECT_NAME} PUBLIC Eigen3::Eigen Ceres::ceres Boost::iostreams protobuf::libprotobuf glog::glog gflags ${LUA_LIBRARIES} ${CAIRO_LIBRARIES} absl::algorithm absl::base absl::debugging absl::flat_hash_map absl::memory absl::meta absl::numeric absl::str_format absl::strings absl::synchronization absl::time absl::utility pthread)
install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME} ARCHIVE DESTINATION lib)
install(DIRECTORY ${VENDOR}/cartographer DESTINATION include FILES_MATCHING PATTERN "*.h")
install(DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/cartographer DESTINATION include FILES_MATCHING PATTERN "*.h")
install(DIRECTORY ${VENDOR}/configuration_files DESTINATION share/${PROJECT_NAME})
install(FILES LICENSE AUTHORS SOURCE_MANIFEST.json DESTINATION share/${PROJECT_NAME})
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET NAMESPACE hector_backend_core::)
ament_export_dependencies(Eigen3 Ceres Boost Protobuf glog gflags absl)
ament_package(CONFIG_EXTRAS cmake/dependencies.cmake)
