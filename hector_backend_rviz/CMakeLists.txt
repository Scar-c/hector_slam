cmake_minimum_required(VERSION 3.16)
project(hector_backend_rviz)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(ament_cmake REQUIRED)
find_package(hector_backend_core REQUIRED)
find_package(hector_backend_msgs REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rviz_rendering REQUIRED)
find_package(rviz_ogre_vendor REQUIRED)
find_package(tf2_ros REQUIRED)
file(GLOB HDRS include/hector_backend_rviz/*.h)
add_library(${PROJECT_NAME} SHARED src/drawable_submap.cpp src/ogre_slice.cpp src/submaps_display.cpp src/submap.cpp ${HDRS})
target_include_directories(${PROJECT_NAME} PUBLIC include ${OGRE_INCLUDE_DIRS})
ament_target_dependencies(${PROJECT_NAME} hector_backend_msgs rclcpp ament_index_cpp rviz_common rviz_rendering rviz_ogre_vendor tf2_ros)
target_link_libraries(${PROJECT_NAME} hector_backend_core::hector_backend_core rviz_common::rviz_common)
pluginlib_export_plugin_description_file(rviz_common rviz_plugin_description.xml)
register_rviz_ogre_media_exports(DIRECTORIES "ogre_media/materials/glsl120" "ogre_media/materials/scripts")
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(FILES LICENSE DESTINATION share/${PROJECT_NAME})
ament_package()
