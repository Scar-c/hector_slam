cmake_minimum_required(VERSION 3.16)
project(hector_global_initialization LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(HECTOR_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_package(ament_cmake REQUIRED)
find_package(hector_backend_core REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgcodecs)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
find_package(nav_msgs REQUIRED)
set(FFTW_PREFIX ${HECTOR_REPO}/3rd/fftw_prefix/usr)
find_path(HECTOR_FFTW_INCLUDE fftw3.h HINTS ${FFTW_PREFIX}/include REQUIRED)
find_library(HECTOR_FFTW_LIBRARY fftw3 HINTS ${FFTW_PREFIX}/lib/x86_64-linux-gnu REQUIRED)
add_library(business_fft_core ${HECTOR_REPO}/3rd/business_fft/src/core.cpp)
target_include_directories(business_fft_core PUBLIC ${HECTOR_REPO}/3rd/business_fft/include ${HECTOR_FFTW_INCLUDE} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(business_fft_core PUBLIC Eigen3::Eigen yaml-cpp Threads::Threads ${OpenCV_LIBS} ${HECTOR_FFTW_LIBRARY})
add_executable(global_initialize global_initialize.cpp)
target_include_directories(global_initialize PRIVATE ${HECTOR_REPO}/hector_mapping/include)
ament_target_dependencies(global_initialize nav_msgs)
target_link_libraries(global_initialize business_fft_core hector_backend_core::hector_backend_core)
