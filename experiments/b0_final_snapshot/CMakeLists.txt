cmake_minimum_required(VERSION 3.16)
project(b0_hector_gravity_cloud LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(hector_mapping REQUIRED)
find_package(zx_carto REQUIRED)
find_package(Boost REQUIRED COMPONENTS iostreams)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
add_executable(hector_cloud_node hector_cloud_node.cpp)
target_include_directories(hector_cloud_node BEFORE PRIVATE include ../../src/zx_slam/zx_carto)

ament_target_dependencies(carto_rotation_deskew rclcpp sensor_msgs tf2_ros geometry_msgs std_srvs std_msgs)
