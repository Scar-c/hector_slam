cmake_minimum_required(VERSION 3.16)
project(hector_cloud_frontend VERSION 0.6.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(hector_mapping REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(hector_backend_core REQUIRED)
set(HECTOR_CARTOGRAPHER_PACKAGE hector_backend_core::hector_backend_core)
foreach(target hector_cloud_mapping_node hector_cloud_localization_node)
  add_executable(${target} src/${target}.cpp)
  target_include_directories(${target} PRIVATE ${hector_mapping_INCLUDE_DIRS})
  target_link_libraries(${target} Eigen3::Eigen yaml-cpp ${HECTOR_CARTOGRAPHER_PACKAGE}
    hector_mapping::hector_mapping__rosidl_typesupport_cpp)
  ament_target_dependencies(${target} hector_mapping rclcpp rcl_interfaces sensor_msgs geometry_msgs nav_msgs std_msgs std_srvs tf2 tf2_ros tf2_geometry_msgs)
endforeach()
add_executable(hector_reference_attitude src/reference_attitude.cpp)
add_executable(hector_carto_deskew src/carto_rotation_deskew.cpp)
foreach(target hector_reference_attitude hector_carto_deskew)
  target_link_libraries(${target} Eigen3::Eigen ${HECTOR_CARTOGRAPHER_PACKAGE})
  ament_target_dependencies(${target} rclcpp sensor_msgs geometry_msgs std_msgs std_srvs tf2_ros tf2_geometry_msgs)
endforeach()
install(TARGETS hector_cloud_mapping_node hector_cloud_localization_node hector_reference_attitude hector_carto_deskew
  DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/seed_pose.py scripts/localization_visualizer.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES LICENSE DESTINATION share/${PROJECT_NAME})
ament_package()
