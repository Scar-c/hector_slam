cmake_minimum_required(VERSION 3.16)
project(hector_backend VERSION 0.6.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(hector_backend_core REQUIRED)
find_package(hector_backend_msgs REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2_ros REQUIRED)
add_executable(hector_online_backend src/online_backend.cpp)
ament_target_dependencies(hector_online_backend hector_backend_msgs ament_index_cpp rclcpp sensor_msgs geometry_msgs nav_msgs std_msgs std_srvs tf2_ros)
target_link_libraries(hector_online_backend hector_backend_core::hector_backend_core)
add_executable(hector_backend_offline src/offline_backend.cpp)
target_link_libraries(hector_backend_offline hector_backend_core::hector_backend_core)
ament_target_dependencies(hector_backend_offline ament_index_cpp)
install(TARGETS hector_online_backend hector_backend_offline DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_package()
